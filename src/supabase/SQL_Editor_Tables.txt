-- Tabla de Categorías de Productos

  id SERIAL PRIMARY KEY,
  name VARCHAR(100) UNIQUE NOT NULL
);

-- Tabla de Categorías de Servicios
CREATE TABLE IF NOT EXISTS service_categories (
  id SERIAL PRIMARY KEY,
  name VARCHAR(100) UNIQUE NOT NULL

-- Tabla de Facturas
CREATE TABLE IF NOT EXISTS invoices (
  id uuid PRIMARY KEY DEFAULT uuid_generate_v4(),
  created_at TIMESTAMPTZ DEFAULT timezone('utc'::text, now()),
  total NUMERIC NOT NULL,
  status TEXT NOT NULL, -- 'pending', 'paid', etc.
  payment_proof TEXT,   -- URL del comprobante
  customer_name TEXT
);

-- Tabla de Items de Factura
CREATE TABLE IF NOT EXISTS invoice_items (
  id SERIAL PRIMARY KEY,
  invoice_id UUID REFERENCES invoices(id) ON DELETE CASCADE,
  product_name TEXT NOT NULL,
  product_type TEXT, -- 'producto' o 'servicio'
  quantity INTEGER NOT NULL,
  price NUMERIC NOT NULL
);

-- Índice para estadísticas de productos/servicios más vendidos
CREATE INDEX IF NOT EXISTS idx_invoice_items_product_name ON invoice_items(product_name);

-- Extensión para UUID
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";
